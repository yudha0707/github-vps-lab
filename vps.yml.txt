name: Linux VPS Lab

on:
  workflow_dispatch:

jobs:
  vps:
    runs-on: ubuntu-latest

    steps:
      - name: System Info
        run: |
          echo "=== SYSTEM ==="
          uname -a
          echo
          echo "=== CPU ==="
          nproc
          echo
          echo "=== RAM ==="
          free -h
          echo
          echo "=== DISK ==="
          df -h

      - name: Install SSH Server
        run: |
          sudo apt update
          sudo apt install -y openssh-server
          sudo systemctl start ssh

      - name: Create lab user
        run: |
          sudo useradd -m -s /bin/bash lab
          echo 'lab:LabPassword123!' | sudo chpasswd
          sudo usermod -aG sudo lab

      - name: Keep runner alive
        run: |
          echo "SSH LAB READY"
          echo "Runner akan aktif sampai job selesai."
          sleep 21600
